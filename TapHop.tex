if (box.exec() != null) {
	var result = String(box.get("LuaChonTapHop"));
	var index = -1;
	
	// Tìm index
	for (var i = 0; i < dsTapHop.length; i++) {
		if (dsTapHop[i] == result) {
			index = i;
			break;
		}
	}
	
	// Chèn code
	if (index > -1) {
		editor.insertSnippet(dataTapHop[index]);
	} else {
		alert("Macro chưa nhận dạng được mục bạn vừa chọn!");

function num(stri){
	stri=stri.replace(/([ \=\t\*])/g,"");
	if (stri[0]=="\(" && stri[stri.length-1]=="\)")
	stri=stri.substring(1,stri.length-1);
	if (stri=="") return 0;
	else if (stri=="+") return 1;
	else if (stri=="-") return -1;
	else if (stri.indexOf("/")>-1) {
		var stri1=Number(stri.substring(0,stri.indexOf("/")));
		var stri2=Number(stri.substring(1+stri.indexOf("/")),stri.length);
		return stri1/stri2;
	}
	else if (stri.length>2 && stri.substring(stri.length-2,stri.length)=="\^2") {
		var stri1=Number(stri.substring(0,stri.indexOf("\^2")));
		return stri1*stri1;
	}
	else if (stri.length>2 && stri.substring(stri.length-2,stri.length)=="\^3") {
		var stri1=Number(stri.substring(0,stri.indexOf("\^3")));
		return stri1*stri1*stri1;
	}
	else if (stri.indexOf("sqrt")>-1) {
		if (stri[0]=="+") stri=stri.substring(1,stri.length);
		if (stri.indexOf("sqrt")==0) {
			var stri1=Number(stri.substring(4,stri.length));
			return Math.sqrt(stri1);
		}
		else if (stri[0]=="-") {
			stri=stri.substring(1,stri.length);
			if (stri.indexOf("sqrt")==0) {
				var stri1=Number(stri.substring(4,stri.length));
				return -Math.sqrt(stri1);
			}
			else {
				var stri1=Number(stri.substring(0,stri.indexOf("sqrt")));
				var stri2=Number(stri.substring(4+stri.indexOf("sqrt"),stri.length));
				return -Math.sqrt(stri1*stri1*stri2);
			}
		}
		else {
			var stri1=Number(stri.substring(0,stri.indexOf("sqrt")));
			var stri2=Number(stri.substring(4+stri.indexOf("sqrt"),stri.length));
			return Math.sqrt(stri1*stri1*stri2);
		}
		
	}
	else return Number(stri);
};
function fracS(num){			//######## Tạo biểu thức dạng String: 'm/n' để tính toán trong Tikz
	for(var n=2;n<49;n++){
		if (Math.round(1000000*num)/1000000!= Math.round(num) && Math.round(1000000*n*num)/1000000==Math.round(n*num)) {
			return Math.round(n*num)+"/"+n;
			break;
		}
		else if(n==48) return Math.round(100*num)/100;
	}
};
function fracN(num){			//######## Tạo biểu thức String dạng '\frac{m}{n}' ghi cho các Node
	for(var n=2;n<49;n++){
		if (Math.round(1000000*num)/1000000!= Math.round(num) && Math.round(1000000*n*num)/1000000==Math.round(n*num)) {
			return "\\frac{"+Math.round(n*num)+"}{"+n+"}";
			break;
		}
		else if(n==48) return Math.round(10*num)/10;
	}
};
function formatS(num){			//######## Tạo biểu thức String dạng 'sqrt n', dùng tính toán trong Tikz
	if (num<0) var Dau="-";
	else Dau="";
	if (Math.round(1000000*num)/1000000==Math.round(num))
	return Dau+Math.round(Math.abs(num));
	else if (Math.round(1000000*num*num)/1000000==Math.round(num*num)) {
		for(var n=40;n>1;n--){
			if (Math.round(1000000*num*num/(n*n))/1000000== Math.round(num*num/(n*n))) {
				return Dau+n+"*sqrt "+Math.round(num*num/(n*n));
				break;
			}
			else if(n==2) return Dau+"sqrt "+Math.round(num*num);
		}
	}
	else 
	for(var n=2;n<41;n++){
		if (Math.round(1000000*num*num*n*n)/1000000== Math.round(num*num*n*n) && Math.round(1000000*Math.sqrt(num*num*n*n))/1000000!= Math.round(Math.sqrt(num*num*n*n))) {
			for(var m=40;m>1;m--){
				if (Math.round(1000000*num*num*n*n/(m*m))/1000000== Math.round(num*num*n*n/(m*m))) {
					return Dau+m+"*sqrt "+Math.round(num*num*n*n/(m*m))+"/"+n;
					break;
				}
				else if(m==2) return Dau+"sqrt "+Math.round(num*num*n*n)+"/"+n;
			}
			break;
		}
		else if(n==40) return Dau+fracS(Math.abs(num));
	}
};

		for(var n=40;n>1;n--){
			if (Math.round(1000000*num*num/(n*n))/1000000== Math.round(num*num/(n*n))) {
				return Dau+n+"\\sqrt{"+Math.round(num*num/(n*n))+"}";
				break;
			}
			else if(n==2) return Dau+"\\sqrt{"+Math.round(num*num)+"}";
		}
	}
	else 
	for(var n=2;n<41;n++){
		if (Math.round(1000000*num*num*n*n)/1000000== Math.round(num*num*n*n) && Math.round(1000000*Math.sqrt(num*num*n*n))/1000000!= Math.round(Math.sqrt(num*num*n*n))) {
			for(var m=40;m>1;m--){
				if (Math.round(1000000*num*num*n*n/(m*m))/1000000== Math.round(num*num*n*n/(m*m))) {
					return Dau+"\\frac{"+m+"\\sqrt{"+Math.round(num*num*n*n/(m*m))+"}}{"+n+"}";
					break;
				}
				else if(m==2) return Dau+"\\frac{"+"\\sqrt{"+Math.round(num*num*n*n)+"}}{"+n+"}";
			}
			break;
		}
		else if(n==40) return Dau+fracN(Math.abs(num));
	}
};

								var xalign="below"; var yalign="above";
								if (yi<0) xalign="above"; 
								if (a<0) yalign="below";
								VeLuoi=VeLuoi+"\n("+formatS(-1*i)+",0)node["+xalign+"]{$"+formatN(-1*i)+"$} |-(0,"+formatS(yi)+")node["+yalign+" left=-1pt]{$"+formatN(yi)+"$} -|("+formatS(i)+",0)node["+xalign+"]{$"+formatN(i)+"$}";
							}
						}
						
						if (VeDiem!="")
						str=str+"\n\\fill"+VeDiem+";";
						if (VeLuoi!="")
						str=str+"\n\\draw[dashed]"+VeLuoi+";";
					}
				}
				
				str=str+"\n\\end{tikzpicture}";
				
				str=str.replace(/([0-9]{1,}\.[0-9]{2})([0-9]{1,})/g, "$1");
			}
			//================= Vẽ đồ thị hàm nhất biến ==============
			
			else if (LoaiHS=="HSNB") {
				if (c==0){alert("Hệ số c phải khác 0 bạn ơi!");}
				else if (a*d==b*c){alert("Bạn nhập hàm số hằng rồi kìa!");}
				else {
					var i=Math.sqrt(Math.abs(a*d-c*b)/(c*c));
					
					var y0=a/c; var x0=-d/c;
					var xmin=Math.round(10*x0-31*i)/10; if(xmin>-1)	xmin=-1;
					var xmax=Math.round(10*x0+31.2*i)/10; if(xmax<1) xmax=1;
					var ymin=Math.round(10*y0-30*i)/10; if(ymin>-1)	ymin=-1;
					var ymax=Math.round(10*y0+30*i)/10; if(ymax<1) 	ymax=1;
					
					var x0t=Math.round(1000*(-d/c-Math.abs((d*ymin-b)/(a-c*ymin)+d/c)))/1000;
					var x0p=Math.round(1000*(-d/c+Math.abs((d*ymin-b)/(a-c*ymin)+d/c)))/1000;
					
					if (i<5){var scal=1;}
					else if (i<6.5){var scal=Math.round(50/i)/10;}
					else if (i<8){var scal=Math.round(65/i)/10;}
					else {var scal=Math.round(80/i)/10;}
					
					var str="\\begin{tikzpicture}[scale="+scal+",font=\\footnotesize,samples=200,>=stealth]\n";
					if (box.get("Oluoi"))
					str=str+"\\draw[color=gray!40,dashed,very thin] ("+xmin+","+ymin+") grid ("+xmax+","+ymax+");\n";
					str=str+"\\draw[->] ("+xmin+",0)--("+xmax+",0) node[below]{$x$};\n";
					str=str+"\\draw[->] (0,"+ymin+")--(0,"+ymax+") node[left]{$y$};\n";
					str=str+"\\node at (0,0) [below left=-2pt]{$O$};\n";
					//str=str+"\\clip ("+xmin+","+ymin+") rectangle ("+xmax+","+ymax+");\n";
					
					str=str+"\\draw plot[domain="+xmin+":"+x0t+"](\\x,{"+Hamso+"});\n";
					str=str+"\\draw plot[domain="+x0p+":"+xmax+"](\\x,{"+Hamso+"});\n";
					if (box.get("Congthuc")) {
						var tu=fx.substring(0,fx.search(/([\)]*\/[x\(])/g)).replace(/([\)\(])/g,"");
						mau=fx.substring(1+fx.search(/(\/[x\(])/g),fx.lenght).replace(/[\)\(]/g,"");
						fx="\\dfrac{"+tu+"}{"+mau+"}";
						str=str+"\\node at ("+formatS(x0)+","+(ymax-1)+")";
						if (a*d>c*b) str=str+"[left=5mm] {$y="+fx+"$};\n";
						else str=str+"[right=5mm] {$y="+fx+"$};\n";
					}
					str=str+"\\draw\n("+formatS(x0)+","+ymin+")--("+formatS(x0)+","+ymax+")\n";
					str=str+"("+xmin+","+formatS(y0)+")--("+xmax+","+formatS(y0)+");";
					if (box.get("DiemDB") && (x0!=0 || y0!=0)) {
						str=str+"\n\\fill";
						if (x0!=0) str=str+"\n("+formatS(x0)+",0) circle("+0.6/scal+"pt) node[below left=-2pt]{$"+formatN(x0)+"$}";
						if (y0!=0) str=str+"\n(0,"+formatS(y0)+") circle("+0.6/scal+"pt) node[above left=-2pt]{$"+formatN(y0)+"$}";
						var x1=500; if (a!=0){x1=-b/a;};
						var y2=500; if (d!=0){y2=b/d;};
						if (x1>xmin && x1<xmax && x1!=0){str=str+"\n("+formatS(x1)+",0) circle("+1.3/scal+"pt) node[below]{$"+formatN(x1)+"$}";}
						if (y2>ymin && y2<ymax && y2!=0){str=str+"\n(0,"+formatS(y2)+") circle("+1.3/scal+"pt) node[left]{$"+formatN(y2)+"$}";}
						str=str+";";
					}
					str=str+"\n\\end{tikzpicture}";
					
					str=str.replace(/([0-9]{1,}\.[0-9]{2})([0-9]{1,})/g, "$1");
				}
			}
			//================= Vẽ đồ thị hàm Bậc 2/Bậc 1 ============
			else if (LoaiHS=="HS21") {
				if (d==0){alert("Hàm số bạn nhập lạ quá!");}
				else if (a*e*e/d+c*d==b*e){alert("Hàm số bạn nhập là hàm số hằng rồi!\n        (tức là tử chia hết cho mẫu)");}
				else {
					var x0=-e/d; var y0=b/d-2*a*e/(d*d);
					var j=4*Math.sqrt(Math.abs(a*a*e*e/(d*d*d*d)+a*c/(d*d)-a*b*e/(d*d*d)));
					var i=j*Math.abs(d/a); if (i<0)i=-i;
					var xmin=Math.round(10*x0-10*i)/10; if(xmin>-1)xmin=-1;
					var xmax=Math.round(10*x0+10*i)/10; if(xmax<1) xmax=1;
					var ymin=Math.round(10*y0-10*j)/10; if(ymin>-1)ymin=-1;
					var ymax=Math.round(10*y0+10*j)/10; if(ymax<1) ymax=1;
					var xtr=xmin;	var ytr=a*xtr/d+b/d-a*e/(d*d); // Dùng vẽ TCX
					var xph=xmax; 	var yph=a*xph/d+b/d-a*e/(d*d); // Dùng vẽ TCX
					var DL2=e*e/(d*d)+c/a-b*e/(a*d);			   // Delta_{y'}/(a^2d^2)
					
					var DTxo=Math.abs(b*b-4*a*c-2*b*d*yph+yph*yph*d*d+4*a*e*yph); // Dùng xác định khoảng nhỏ gần TCĐ
					var k=Math.abs(Math.sqrt(DTxo)/Math.abs(2*a)-xph/2+x0/2);
					var x0t=Math.round(100*x0-100*k)/100;
					var x0p=Math.round(100*x0+100*k)/100;
					
					if (i<5){var scal=1;}
					else if (i<6.5){var scal=Math.round(50/i)/10;}
					else if (i<8){var scal=Math.round(65/i)/10;}
					else {var scal=Math.round(80/i)/10;}
					
					var str="\\begin{tikzpicture}[scale="+scal+",font=\\footnotesize,samples=200,>=stealth]\n";
					if (box.get("Oluoi"))
					str=str+"\\draw[color=gray!40,dashed,very thin] ("+xmin+","+ymin+") grid ("+xmax+","+ymax+");\n";
					str=str+"\\draw[->] ("+xmin+",0)--("+xmax+",0) node[below]{$x$};\n";
					str=str+"\\draw[->] (0,"+ymin+")--(0,"+ymax+") node[left]{$y$};\n";
					str=str+"\\node at (0,0) [below left=-2pt]{$O$};";
					// str=str+"\n\\clip ("+xmin+","+ymin+") rectangle ("+xmax+","+ymax+");";
					if (DL2<0) {
						str=str+"\n\\draw plot[domain="+xmin+":"+x0t+"](\\x,{"+Hamso+"});";
						str=str+"\n\\draw plot[domain="+x0p+":"+xmax+"](\\x,{"+Hamso+"});";
					}
					else {
						str=str+"\n\\draw plot[domain="+Math.round(10*xmin+10*k)/10+":"+x0t+"](\\x,{"+Hamso+"});";
						str=str+"\n\\draw plot[domain="+x0p+":"+Math.round(10*xmax-10*k)/10+"](\\x,{"+Hamso+"});";
					}
					if (box.get("Congthuc")) {
						var tu=fx.substring(0,fx.search(/(\/[x\(])/g)).replace(/([\)\(])/g,"");
						mau=fx.substring(1+fx.search(/(\/[x\(])/g),fx.lenght).replace(/[\)\(]/g,"");
						fx="\\dfrac{"+tu+"}{"+mau+"}";
						str=str+"\n\\node at ("+xmin+","+(y0+1)+")[right]{$y="+fx+"$};";
						
						var ag=a/d; var bg=b/d-a*e/(d*d);
						if (ag==1){ag="x";}
						else if (ag==-1){ag="-x";}
						else {ag=""+formatN(ag)+"x";}
						if (bg==0){bg="";}
						else if(bg>0){bg="+"+formatN(bg)+"";}
						else {bg=""+formatN(bg)+"";}
					}
					
					str=str+"\n\\draw\n("+formatS(x0)+","+ymin+")--("+formatS(x0)+","+ymax+")";
					str=str+"\n("+formatS(xtr)+","+formatS(ytr)+")--("+formatS(xph)+","+formatS(yph)+")";
					if (a*e*e/d+c*d>b*e && box.get("Congthuc")) 
					str=str+" node[pos=0.85,below,sloped]{$y="+ag+bg+"$}";
					else if (a*e*e/d+c*d<b*e && box.get("Congthuc")) 
					str=str+" node[pos=0.85,above,sloped]{$y="+ag+bg+"$}";
					str=str+";";
					
					if (box.get("DiemDB")) {
						if (DL2>0){
							str=str+"\n\\fill";
							var x1=x0-Math.sqrt(DL2); var y1=y0-Math.sqrt(DL2)*2*a/d;
							var x2=x0+Math.sqrt(DL2); var y2=y0+Math.sqrt(DL2)*2*a/d;
							str=str+"\n("+formatS(x1)+","+formatS(y1)+") circle("+1.3/scal+"pt)";
							str=str+"\n("+formatS(x2)+","+formatS(y2)+") circle("+1.3/scal+"pt)";
							if (b*b-4*a*c>0) {
								var n1=(-b-Math.sqrt(b*b-4*a*c))/(2*a);
								var n2=(-b+Math.sqrt(b*b-4*a*c))/(2*a);
								str=str+"\n("+formatS(n1)+",0) circle("+1.3/scal+"pt)";
								str=str+"\n("+formatS(n2)+",0) circle("+1.3/scal+"pt)";
							}
							str=str+";";
							str=str+"\n\\draw[dashed]";
							if (x1*y1 != 0) {
								if(x1<0 && y1<0) {str=str+"\n("+formatS(x1)+",0)node[above]{$"+formatN(x1)+"$} |-(0,"+formatS(y1)+")node[right]{$"+formatN(y1)+"$}";}
								else if(x1<0 && y1>0){str=str+"\n("+formatS(x1)+",0)node[below]{$"+formatN(x1)+"$} |-(0,"+formatS(y1)+")node[right]{$"+formatN(y1)+"$}";}
								else if(x1>0 && y1<0){str=str+"\n("+formatS(x1)+",0)node[above]{$"+formatN(x1)+"$} |-(0,"+formatS(y1)+")node[left]{$"+formatN(y1)+"$}";}
								else 				 {str=str+"\n("+formatS(x1)+",0)node[below]{$"+formatN(x1)+"$} |-(0,"+formatS(y1)+")node[left]{$"+formatN(y1)+"$}";}
							}
							if (x2*y2 != 0) {
								if(x2<0 && y2<0) {str=str+"\n("+formatS(x2)+",0)node[above]{$"+formatN(x2)+"$} |-(0,"+formatS(y2)+")node[right]{$"+formatN(y2)+"$}";}
								else if(x2<0 && y2>0){str=str+"\n("+formatS(x2)+",0)node[below]{$"+formatN(x2)+"$} |-(0,"+formatS(y2)+")node[right]{$"+formatN(y2)+"$}";}
								else if(x2>0 && y2<0){str=str+"\n("+formatS(x2)+",0)node[above]{$"+formatN(x2)+"$} |-(0,"+formatS(y2)+")node[left]{$"+formatN(y2)+"$}";}
								else 				 {str=str+"\n("+formatS(x2)+",0)node[below]{$"+formatN(x2)+"$} |-(0,"+formatS(y2)+")node[left]{$"+formatN(y2)+"$}";}
							}
							str=str+";";
						}
						else {
							str=str+"\n\\fill";
							if (b*b-4*a*c>0) {
								var n1=(-b-Math.sqrt(b*b-4*a*c))/(2*a);
								var n2=(-b+Math.sqrt(b*b-4*a*c))/(2*a);
								str=str+"\n("+formatS(n1)+",0) circle("+1.3/scal+"pt)";
								str=str+"\n("+formatS(n2)+",0) circle("+1.3/scal+"pt)";
							}
							str=str+";";
						}
					}
					str=str+"\n\\end{tikzpicture}";
					
					str=str.replace(/([0-9]{1,}\.[0-9]{2})([0-9]{1,})/g, "$1");
				}
			}
			//========================================================
			editor.insertSnippet(str);
		}
		else alert(strHelp);
	}
